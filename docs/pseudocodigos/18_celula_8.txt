FUNÇÃO _clean(f)
    f ← (f.filled(0.0) SE isinstance(f, np.ma.MaskedArray) SENÃO np.nan_to_num(f, nan=0.0))
    RETORNAR np.clip(f, 0.0, 205.0).astype(np.float64)

FUNÇÃO _objects(field)
    // Objetos contíguos acima de R95/15: centro de massa global, dispersão r e volume escalado V.
    wet ← field[field ≥ SAL_MIN_RAIN]
    SE wet.size = 0 ENTÃO
        RETORNAR NULO
    FIM SE
    thr ← max(SAL_MIN_RAIN, np.percentile(wet, 95) × SAL_OBJ_FRACTION)
    (lab, n) ← label(field ≥ thr)
    cm_glob ← np.array(center_of_mass(field))
    num_r ← num_v ← mass ← 0.0
    PARA CADA k EM range(1, n + 1) FAÇA
        m ← lab = k
        r_n ← float(field[m].sum())
        v_n ← r_n / float(field[m].max())
        d_n ← float(np.linalg.norm(np.array(center_of_mass(field × m)) − cm_glob))
        num_r ← num_r + r_n × d_n
        num_v ← num_v + r_n × v_n
        mass ← mass + r_n
    FIM PARA
    RETORNAR {"cm": cm_glob, "r": num_r / mass, "V": num_v / mass, "n_obj": n}
FIM FUNÇÃO
FUNÇÃO compute_sal_step(obs_2d, mod_2d, res_km, min_rain_thr)
    // S, A, L (Wernli et al., 2008) e deslocamento do centro de massa em km.
    // 
    // SAL não é definido quando um dos campos não tem chuva: devolve NaN e o motivo.
    (R_o, R_m) ← (_clean(obs_2d), _clean(mod_2d))
    out ← {"S": np.nan, "A": np.nan, "L": np.nan, "SAL_abs": np.nan, "Shift_km": np.nan, "Razao_Volume": np.nan, "N_obj_obs": 0, "N_obj_mod": 0, "Sta…
    (o, m) ← (_objects(R_o), _objects(R_m))
    SE o É NULO OU m É NULO ENTÃO
        out["Status"] ← "indefinido: " + ("sem chuva observada" SE o É NULO SENÃO "sem chuva prevista")
        RETORNAR out
    FIM SE
    (h, w) ← R_o.shape
    d_max ← float(np.hypot(h − 1, w − 1))
    (mo, mm) ← (R_o.mean(), R_m.mean())
    A ← mm − mo / 0.5 × mm + mo
    S ← m["V"] − o["V"] / 0.5 × m["V"] + o["V"]
    shift_px ← float(np.linalg.norm(m["cm"] − o["cm"]))
    L ← min(2.0, shift_px / d_max + 2.0 × abs(m["r"] − o["r"]) / d_max)
    EXECUTAR out.update({"S": round(float(S), 4), "A": round(float(A), 4), "L": round(float(L), 4), "SAL_abs": round(float(abs(S) + abs(A) + L), 4), …
    RETORNAR out

FUNÇÃO sal_over_set(obs_seqs, preds)
    // SAL em muitas janelas (ex.: teste 2025 na Célula 12): mediana e intervalo interquartil por modelo
    // e horizonte, mais a contagem de casos indefinidos. obs_seqs: (N, 4, H, W); preds: nome -> (N, 4, H, W).
    rows ← []
    PARA CADA (name, P) EM preds.items() FAÇA
        PARA CADA s EM range(obs_seqs.shape[1]) FAÇA
            r ← pd.DataFrame([compute_sal_step(obs_seqs[(i, s)], P[(i, s)]) PARA CADA i EM range(obs_seqs.shape[0])])
            ok ← r[r["Status"] = "ok"]
            row ← {"Modelo": name, "Lead Time": f'+{30 * (s + 1)} min', "Janelas": len(r), "Indefinidas": int(len(r) − len(ok))}
            PARA CADA c EM ("S", "A", "L", "Shift_km", "SAL_abs", "Razao_Volume") FAÇA
                q ← (ok[c].quantile([0.25, 0.5, 0.75]) SE len(ok) SENÃO pd.Series([np.nan] × 3, index=[0.25, 0.5, 0.75]))
                row[f'{c} mediana'] ← round(float(q[0.5]), 3)
                row[f'{c} IQR'] ← f'{q[0.25]:.2f} a {q[0.75]:.2f}'
            FIM PARA
            EXECUTAR rows.append(row)
        FIM PARA
    FIM PARA
    RETORNAR pd.DataFrame(rows)
