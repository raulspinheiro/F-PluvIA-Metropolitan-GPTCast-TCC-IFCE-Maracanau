FUNÇÃO compute_frame_rapsd(field_2d, res_km)
    // RAPSD de um campo (mm/h). Devolve comprimentos de onda (km) e potência, sem a componente DC.
    f ← (field_2d.filled(0.0) SE isinstance(field_2d, np.ma.MaskedArray) SENÃO np.nan_to_num(field_2d, nan=0.0))
    (psd, freq) ← spectral.rapsd(np.clip(f, 0.0, 205.0).astype(np.float64), fft_method=np.fft, return_freq=VERDADEIRO, d=res_km, normalize=FALSO)
    ok ← freq > 0
    RETORNAR (1.0 / freq[ok], psd[ok])
FIM FUNÇÃO
FUNÇÃO band_ratios(wl, psd_model, psd_obs)
    // Energia: soma da potência na banda (modelo / observado). Forma: o mesmo com cada espectro
    // dividido pela sua potência total, isolando a distribuição entre escalas da amplitude do campo.
    out ← {}
    (tot_m, tot_o) ← (psd_model.sum(), psd_obs.sum())
    PARA CADA (name, (lo, hi)) EM BANDS.items() FAÇA
        m ← wl ≥ lo & wl < hi
        e_o ← psd_obs[m].sum()
        out[f'Energia {name}'] ← (round(float(psd_model[m].sum() / e_o), 3) SE e_o > 0 SENÃO np.nan)
        out[f'Forma {name}'] ← (round(float(psd_model[m].sum() / tot_m / e_o / tot_o), 3) SE e_o > 0 E tot_m > 0 SENÃO np.nan)
    FIM PARA
    RETORNAR out
FIM FUNÇÃO
