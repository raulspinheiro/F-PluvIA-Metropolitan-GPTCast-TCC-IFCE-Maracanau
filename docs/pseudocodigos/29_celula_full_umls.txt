FUNÇÃO sem_magicas(src)
    // Converte comandos do Colab (!pip, %cd, x = !ls) em Python puro, como o próprio IPython faz.
    TENTAR
        IMPORTAR TransformerManager DE IPython.core.inputtransformer2
        RETORNAR TransformerManager().transform_cell(src)
    SE OCORRER ERRO Exception
        (nada a fazer)
    FIM TENTAR
    (linhas, aceitas) ← ([], [])   // alternativa: comando só fora de parênteses e de textos
    FUNÇÃO dentro_de_expressao(texto)
        prof ← 0
        TENTAR
            PARA CADA tok EM tokenize.generate_tokens(io.StringIO(texto).readline) FAÇA
                SE tok.type = tokenize.OP E tok.string EM "([{" ENTÃO
                    prof ← prof + 1
                SENÃO SE tok.type = tokenize.OP E tok.string EM ")]}" ENTÃO
                    prof ← prof − 1
                FIM SE
            FIM PARA
        SE OCORRER ERRO tokenize.TokenError (chamado e)
            RETORNAR "string" EM str(e) OU prof > 0
        SE OCORRER ERRO (IndentationError, SyntaxError)
            (nada a fazer)
        FIM TENTAR
        RETORNAR prof > 0
    FIM FUNÇÃO
    PARA CADA ln EM src.split(" ") FAÇA
        t ← ln.lstrip()
        SE (t.startswith(("!", "%")) OU re.match("^[\w\s,]+=\s*!", t)) E NÃO dentro_de_expressao(" ".join(aceitas) + " ") ENTÃO
            EXECUTAR linhas.append(ln[:len(ln) − len(t)] + "pass  # " + t)
        SENÃO
            EXECUTAR linhas.append(ln)
            EXECUTAR aceitas.append(ln)
        FIM SE
    FIM PARA
    RETORNAR " ".join(linhas)
