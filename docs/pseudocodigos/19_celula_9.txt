    rows ← []

        row ← {"Ambiente_Execucao": ENV_LABEL, "Hardware_Tag": ENV_TAG, "Modelo": model_name, "Lead Time": f'+{(i + 1) * step_minutes} min', "Lead_mi…
        PARA CADA thr EM thresholds FAÇA
            (ob, pb) ← (o ≥ thr, p ≥ thr)
            (tp, fp, fn) ← (int(np.sum(ob & pb)), int(np.sum(~ob & pb)), int(np.sum(ob & ~pb)))
            csi ← (tp / tp + fp + fn SE tp + fp + fn SENÃO np.nan)
            bias ← (tp + fp / tp + fn SE tp + fn SENÃO np.nan)
            row[f'CSI_{thr}'] ← (round(csi, 4) SE np.isfinite(csi) SENÃO np.nan)
            row[f'BIAS_{thr}'] ← (round(bias, 3) SE np.isfinite(bias) SENÃO np.nan)
            SE NÃO np.isfinite(csi) ENTÃO
                row[f'CSI/BIAS @ {thr} mm/h'] ← "sem eventos"
            SENÃO SE NÃO np.isfinite(bias) ENTÃO
                row[f'CSI/BIAS @ {thr} mm/h'] ← f'{csi:.3f} / sem obs'
            SENÃO
                row[f'CSI/BIAS @ {thr} mm/h'] ← f'{csi:.3f} / {bias:.2f}'
            FIM SE
        FIM PARA
        EXECUTAR rows.append(row)
    FIM PARA
    RETORNAR pd.DataFrame(rows)
