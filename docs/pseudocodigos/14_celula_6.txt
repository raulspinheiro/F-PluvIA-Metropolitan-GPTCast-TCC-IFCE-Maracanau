FUNÇÃO reconstruct_sequence_vqgan(model_gptcast, seq_dbz, device, batch_frames)
    // (S, H, W) em dBZ [0, 60] -> codificador -> codebook (1.024 códigos, 8 x 8 tokens) -> decodificador.
    // 
    // Devolve (reconstrução em dBZ, em mm/h, tempo de GPU em s), processando em lotes.
    vae ← model_gptcast.first_stage_model.to(device)
    EXECUTAR vae.eval()
    vdtype ← next(vae.parameters()).dtype
    raw ← (seq_dbz.filled(0.0) SE isinstance(seq_dbz, np.ma.MaskedArray) SENÃO np.nan_to_num(seq_dbz, nan=0.0))
    raw ← np.clip(raw, 0.0, 60.0).astype(np.float32)
    out ← np.empty_like(raw)
    t0 ← _sync()
    PARA CADA i EM range(0, raw.shape[0], batch_frames) FAÇA
        x ← torch.from_numpy(raw[(i:i + batch_frames, NULO)] / 30.0 − 1.0).to(device=device, dtype=vdtype)
        (quant, _, _) ← vae.encode(x)
        out[i:i + batch_frames] ← vae.decode(quant).squeeze(1).float().cpu().numpy()
    FIM PARA
    t_gpu ← _sync() − t0
    rec_dbz ← np.clip(out + 1.0 × 30.0, 0.0, 60.0).astype(np.float32)
    RETORNAR (rec_dbz, _to_rain(rec_dbz), float(t_gpu))
FIM FUNÇÃO

FUNÇÃO compute_stage1_metrics(obs_rain, rec_rain, obs_dbz, rec_dbz, gpu_time_s, label)
    // MAE/RMSE (mm/h e dBZ), SSIM em dBZ (faixa fixa de 60 dBZ), PCC, volume, picos, CSI/BIAS.
    (o_r, p_r) ← (obs_rain.astype(np.float64), rec_rain.astype(np.float64))
    (o_d, p_d) ← (obs_dbz.astype(np.float64), rec_dbz.astype(np.float64))
    ssim_d ← float(np.mean([ssim(o_d[i], p_d[i], data_range=60.0) PARA CADA i EM range(o_d.shape[0])]))
    pcc ← (float(np.corrcoef(o_r.ravel(), p_r.ravel())[(0, 1)]) SE o_r.std() > 1e-06 E p_r.std() > 1e-06 SENÃO np.nan)
    row ← {"Ambiente_Execucao": ENV_LABEL, "Hardware_Tag": ENV_TAG, "Conjunto Avaliado": label, "Quadros Avaliados": int(o_r.shape[0]), "MAE [mm/h] (…
    PARA CADA thr EM THRESHOLDS FAÇA
        (ob, rb) ← (o_r ≥ thr, p_r ≥ thr)
        (tp, fp, fn) ← (int(np.sum(ob & rb)), int(np.sum(~ob & rb)), int(np.sum(ob & ~rb)))
        csi ← (tp / tp + fp + fn SE tp + fp + fn SENÃO np.nan)
        bias ← (tp + fp / tp + fn SE tp + fn SENÃO np.nan)
        row[f'CSI/BIAS @ {thr} mm/h'] ← (f'{csi:.3f} / {bias:.2f}' SE np.isfinite(csi) SENÃO "sem eventos")
    FIM PARA
    row["Tempo Reconstrução GPU [s] (↓)"] ← round(float(gpu_time_s), 4)
    row["ms por Quadro (↓)"] ← round(1000 × gpu_time_s / max(o_r.shape[0], 1), 3)
    RETORNAR row
FIM FUNÇÃO

FUNÇÃO unique_test_frames(dataset)
    // As janelas de teste se sobrepõem (passo de 60 min, 11 quadros de 30 min): cada horário aparece
    // em várias janelas. Devolve cada quadro de 30 min uma única vez, com o seu horário.
    (S, T) ← (dataset.sequences_dbz, dataset.timestamps)
    n_f ← S.shape[1]
    offsets ← np.array([k − n_f − 1 × STEP_MIN PARA CADA k EM range(n_f)], dtype="timedelta64[m]")
    times ← T[(:, NULO)] + offsets[(NULO, :)].ravel()
    (_, first) ← np.unique(times, return_index=VERDADEIRO)
    (seq_idx, frame_idx) ← np.divmod(first, n_f)
    RETORNAR (np.clip(S[(seq_idx, frame_idx)], 0.0, 60.0).astype(np.float32), times[first])
