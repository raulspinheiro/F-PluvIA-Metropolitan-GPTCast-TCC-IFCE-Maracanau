FUNÇÃO _lote(model, n)
    // O pacote grava cada token com predicted_indices[:, i, j] = res, e res vem com forma (lote, 1): com lote > 1 isso
    // falha. Só durante a geração em lote, o predict_next_index() deste modelo devolve (lote,); depois volta ao original.
    SE n = 1 ENTÃO
        EXECUTAR (yield)
        RETORNAR
    FIM SE
    original ← model.predict_next_index
    model.predict_next_index ← FUNÇÃO(contexto, temperature, top_k) ↦ original(contexto, temperature, top_k)[(:, 0)]
    TENTAR
        EXECUTAR (yield)
    AO FINAL, SEMPRE
        REMOVER model.predict_next_index
    FIM TENTAR

FUNÇÃO _cont(o, p, thr)
    (ob, pb) ← (o ≥ thr, p ≥ thr)
    tp ← float(np.sum(ob & pb))
    fp ← float(np.sum(~ob & pb))
    fn ← float(np.sum(ob & ~pb))
    tn ← float(ob.size) − tp − fp − fn
    RETORNAR (tp, fp, fn, tn)
FIM FUNÇÃO
FUNÇÃO _scores(tp, fp, fn, tn)
    n ← tp + fp + fn + tn
    csi ← (tp / tp + fp + fn SE tp + fp + fn SENÃO np.nan)
    bias ← (tp + fp / tp + fn SE tp + fn SENÃO np.nan)
    pod ← (tp / tp + fn SE tp + fn SENÃO np.nan)
    far ← (fp / tp + fp SE tp + fp SENÃO np.nan)
    hss_den ← tp + fn × fn + tn + tp + fp × fp + tn
    hss ← (2 × tp × tn − fp × fn / hss_den SE hss_den SENÃO np.nan)
    tpr ← (tp + fn × tp + fp / n SE n SENÃO np.nan)
    ets ← (tp − tpr / tp + fp + fn − tpr SE n E tp + fp + fn − tpr SENÃO np.nan)
    RETORNAR (csi, bias, pod, far, hss, ets)
